cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" D2SMM_VERSION)
string(STRIP "${D2SMM_VERSION}" D2SMM_VERSION)
project(d2smm VERSION ${D2SMM_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The JSON dependency is vendored, licensed, and pinned; no network at configure.
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/nlohmann/json.hpp" json_sha)
if(NOT json_sha STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "Vendored nlohmann/json 3.12.0 differs from its pinned source.")
endif()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
message(STATUS "Build-time crypto: ${OPENSSL_CRYPTO_LIBRARY} (${OPENSSL_VERSION})")
add_executable(d2smm-build tools/build.cpp)
target_include_directories(d2smm-build PRIVATE vendor)
target_link_libraries(d2smm-build PRIVATE OpenSSL::Crypto)
target_compile_options(d2smm-build PRIVATE -Wall -Wextra -Wpedantic)

add_executable(d2smm-media tools/media.cpp)
target_include_directories(d2smm-media PRIVATE vendor)
target_compile_options(d2smm-media PRIVATE -Wall -Wextra -Wpedantic)

if(APPLE)
  enable_language(OBJCXX)
  add_executable(d2smm-cdp tools/cdp.mm tools/cdp_client.mm tools/cdp_record.mm)
  target_compile_options(d2smm-cdp PRIVATE -fobjc-arc)
  target_link_libraries(d2smm-cdp PRIVATE "-framework Foundation")
endif()

enable_testing()
add_subdirectory(mods/terrorized/core)
target_link_libraries(d2smm-build PRIVATE d2smm-terrorized-core)
add_test(NAME build-tool-self-check COMMAND d2smm-build --self-check)
add_test(NAME source-links COMMAND d2smm-build --check-source "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME build-tool-version COMMAND d2smm-build --version)
add_test(NAME media-tool-version COMMAND d2smm-media --version)
set_tests_properties(media-tool-version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "d2smm-media 1.0.0")
set_tests_properties(build-tool-version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "d2smm-build 1.0.0")
if(APPLE)
  add_test(NAME cdp-version COMMAND d2smm-cdp --version)
  set_tests_properties(cdp-version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "d2-debug-cdp r5")
endif()
